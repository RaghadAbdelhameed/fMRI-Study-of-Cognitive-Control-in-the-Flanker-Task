#!/bin/bash

# Loop through cope 1 (Incongruent) and cope 2 (Congruent)
for cope_num in 1 2; do

  # Assign a readable name for the terminal output
  if [ $cope_num -eq 1 ]; then
    condition_name="Incongruent"
  elif [ $cope_num -eq 2 ]; then
    condition_name="Congruent"
  fi

  echo "======================================================="
  echo "Preparing FLAME 1 Cluster analysis for Cope ${cope_num} (${condition_name})..."

  # 1. Swaps 'cope3.feat' for 'cope1.feat' or 'cope2.feat' in the input paths
  # 2. Enforces FLAME 1 statistical model (mixed_yn = 2)
  # 3. Enforces Cluster thresholding (thresh = 3)
  # 4. Updates the output directory name so they have their own folders
  sed -e "s/cope3\.feat/cope${cope_num}\.feat/g" \
      -e "s/set fmri(mixed_yn) .*/set fmri(mixed_yn) 2/g" \
      -e "s/set fmri(thresh) .*/set fmri(thresh) 3/g" \
      -e "s/Flanker_Level3_cope3_master/Flanker_Level3_cope${cope_num}_FLAME1_Cluster/g" \
      design_level3_cope3_master.fsf > design_level3_cope${cope_num}_FLAME1_Cluster.fsf

  echo "Starting FEAT for Cope ${cope_num} (${condition_name})..."

  # Run the newly generated design file
  feat design_level3_cope${cope_num}_FLAME1_Cluster.fsf

done

echo "======================================================="
echo "FLAME 1 Cluster analyses for Cope 1 and Cope 2 have been successfully launched!"
